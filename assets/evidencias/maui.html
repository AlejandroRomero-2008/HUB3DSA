MAUI · cálculo de sueldo por horas

Página móvil real que lee horas y pago por hora, calcula el sueldo y muestra el resultado con DisplayAlert.

Extracto del archivo original

namespace SueldoHorasESTUDIANTE;

public partial class Sueldo : ContentPage
{
	public Sueldo()
	{
		InitializeComponent();
	}

    private async void Button_Clicked(object sender, EventArgs e)
    {
		double hora = Convert.ToDouble(Horas.Text);
		double pagohora = Convert.ToDouble(Pagohora.Text);

		double resul = hora * pagohora;

		await DisplayAlert("Resultado",$"Tu ganaste en tu semana: {resul}","OK");
    }
}
